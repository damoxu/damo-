# build.py
# 作用：把 content/ 里的 Markdown 文件，转成 output/ 里的 HTML 文件

import os
import shutil
import markdown
from jinja2 import Environment, FileSystemLoader

# ─────────────────────────────────────────
# 1. 配置：整个项目只有这里需要改
# ─────────────────────────────────────────

BLOG_TITLE  = "我的博客"
AUTHOR_NAME = "damoxu"

DIR_CONTENT   = "content"    # Markdown 文章放这里
DIR_TEMPLATES = "templates"  # HTML 模板放这里
DIR_STATIC    = "static"     # CSS / 图片放这里
DIR_OUTPUT    = "output"     # 生成结果放这里


# ─────────────────────────────────────────
# 2. 工具函数：每个函数只做一件事
# ─────────────────────────────────────────

def read_markdown_file(filepath):
    """
    读取一个 Markdown 文件，返回：
    - meta : 文章头部信息（标题、日期、标签）
    - html : 正文转成的 HTML 字符串
    """
    with open(filepath, encoding="utf-8") as f:
        raw_text = f.read()

    md = markdown.Markdown(extensions=["meta", "fenced_code", "tables"])
    html_body = md.convert(raw_text)
    meta = {key: values[0] for key, values in md.Meta.items()}
    return meta, html_body


def collect_all_posts(posts_dir):
    posts = []
    for filename in sorted(os.listdir(posts_dir), reverse=True):
        if not filename.endswith(".md"):
            continue
        filepath = os.path.join(posts_dir, filename)
        meta, html_body = read_markdown_file(filepath)
        post = {
            "slug"  : filename.replace(".md", ""),
            "title" : meta.get("title", "无标题"),
            "date"  : meta.get("date",  "未知日期"),
            "tags"  : meta.get("tags",  "").split(","),
            "body"  : html_body,
            "url"   : f"/damo-/posts/{filename.replace('.md', '.html')}",
        }
        posts.append(post)
    return posts


def collect_all_works(works_dir):
    works = []
    for filename in sorted(os.listdir(works_dir)):
        if not filename.endswith(".md"):
            continue
        filepath = os.path.join(works_dir, filename)
        meta, html_body = read_markdown_file(filepath)
        work = {
            "slug"  : filename.replace(".md", ""),
            "title" : meta.get("title", "无标题"),
            "desc"  : meta.get("desc",  ""),
            "link"  : meta.get("link",  "#"),
            "body"  : html_body,
            "url"   : f"/damo-/works/{filename.replace('.md', '.html')}",
        }
        works.append(work)
    return works


def setup_jinja(templates_dir):
    return Environment(loader=FileSystemLoader(templates_dir))


def render_page(env, template_name, output_path, **data):
    template = env.get_template(template_name)
    html = template.render(**data, blog_title=BLOG_TITLE, author=AUTHOR_NAME)
    os.makedirs(os.path.dirname(output_path), exist_ok=True)
    with open(output_path, "w", encoding="utf-8") as f:
        f.write(html)
    print(f"  ✓ 生成：{output_path}")


def clean_output_dir():
    if os.path.exists(DIR_OUTPUT):
        shutil.rmtree(DIR_OUTPUT)
    os.makedirs(DIR_OUTPUT)


def copy_static_files():
    src = DIR_STATIC
    dst = os.path.join(DIR_OUTPUT, "static")
    shutil.copytree(src, dst)
    print(f"  ✓ 复制静态文件：{src} → {dst}")


def build_search_index(posts, works, output_dir):
    import json
    import re
    index = []
    for post in posts:
        plain_text = re.sub(r"<[^>]+>", "", post["body"])
        index.append({
            "title" : post["title"],
            "date"  : post["date"],
            "tags"  : post["tags"],
            "url"   : post["url"],
            "body"  : plain_text[:500],
            "type"  : "post",
        })
    for work in works:
        plain_text = re.sub(r"<[^>]+>", "", work["body"])
        index.append({
            "title" : work["title"],
            "desc"  : work["desc"],
            "url"   : work["url"],
            "body"  : plain_text[:500],
            "type"  : "work",
        })
    index_path = os.path.join(output_dir, "search-index.json")
    with open(index_path, "w", encoding="utf-8") as f:
        json.dump(index, f, ensure_ascii=False, indent=2)
    print(f"  ✓ 生成搜索索引：{index_path}")


def build():
    print("\n🔨 开始构建...\n")
    clean_output_dir()
    copy_static_files()
    posts = collect_all_posts(os.path.join(DIR_CONTENT, "posts"))
    works = collect_all_works(os.path.join(DIR_CONTENT, "works"))
    env = setup_jinja(DIR_TEMPLATES)

    print("\n📝 生成文章页面：")
    for post in posts:
        output_path = os.path.join(DIR_OUTPUT, post["url"].lstrip("/"))
        render_page(env, "post.html", output_path, post=post)

    print("\n🎨 生成作品页面：")
    for work in works:
        output_path = os.path.join(DIR_OUTPUT, work["url"].lstrip("/"))
        render_page(env, "work.html", output_path, work=work)

    print("\n🏠 生成首页：")
    render_page(env, "index.html",
                os.path.join(DIR_OUTPUT, "index.html"),
                posts=posts, works=works)

    print("\n🔍 生成搜索索引：")
    build_search_index(posts, works, DIR_OUTPUT)

    print("\n🔍 生成搜索页：")
    render_page(env, "search.html",
                os.path.join(DIR_OUTPUT, "search.html"),
                posts=posts, works=works)

    print("\n✅ 构建完成！\n")


if __name__ == "__main__":
    build()

const searchInput   = document.getElementById("search-input");
const searchResults = document.getElementById("search-results");
const searchCount   = document.getElementById("search-count");
let searchIndex = [];

fetch("/damo-/search-index.json")
  .then(response => response.json())
  .then(data => {
    searchIndex = data;
    const urlParams = new URLSearchParams(window.location.search);
    const query = urlParams.get("q") || "";
    if (query) {
      searchInput.value = query;
      doSearch(query);
    }
  });

searchInput.addEventListener("input", () => {
  doSearch(searchInput.value.trim());
});

function doSearch(query) {
  if (!query) {
    searchResults.innerHTML = "";
    searchCount.textContent = "";
    return;
  }
  const keyword = query.toLowerCase();
  const matched = searchIndex.filter(item => {
    const inTitle = item.title.toLowerCase().includes(keyword);
    const inBody  = item.body.toLowerCase().includes(keyword);
    const inTags  = (item.tags || []).join(" ").toLowerCase().includes(keyword);
    return inTitle || inBody || inTags;
  });
  renderResults(matched, query);
}

function renderResults(items, query) {
  searchCount.textContent = `找到 ${items.length} 个结果`;
  if (items.length === 0) {
    searchResults.innerHTML = `<li style="padding: 1rem 0; color: var(--color-text-muted)">没有找到关于「${query}」的内容</li>`;
    return;
  }
  searchResults.innerHTML = items.map(item => `
    <li class="post-list-item">
      <a href="${item.url}" class="post-list-title">${highlight(item.title, query)}</a>
      <span class="post-list-date">${item.type === "post" ? item.date : "作品"}</span>
    </li>
  `).join("");
}

function highlight(text, query) {
  const regex = new RegExp(`(${query})`, "gi");
  return text.replace(regex, "<mark>$1</mark>");
}

# serve.py
# 使用方式： ```python serve.py

import os
import time
import threading
import http.server
import webbrowser
from build import build

PORT         = 8000
WATCH_DIRS   = ["content", "templates", "static"]
CHECK_INTERVAL = 1

def get_all_files_with_mtime(dirs):
    files = {}
    for directory in dirs:
        if not os.path.exists(directory):
            continue
        for root, _, filenames in os.walk(directory):
            for filename in filenames:
                filepath = os.path.join(root, filename)
                files[filepath] = os.path.getmtime(filepath)
    return files

def has_any_file_changed(old, new):
    if set(old.keys()) != set(new.keys()):
        return True
    for filepath, mtime in new.items():
        if old.get(filepath) != mtime:
            return True
    return False

def watch_and_rebuild():
    print(f"👀 监听文件变化中：{WATCH_DIRS}")
    snapshot = get_all_files_with_mtime(WATCH_DIRS)
    while True:
        time.sleep(CHECK_INTERVAL)
        new_snapshot = get_all_files_with_mtime(WATCH_DIRS)
        if has_any_file_changed(snapshot, new_snapshot):
            print("\n📄 检测到文件变化，重新构建...")
            build()
            snapshot = new_snapshot

class QuietHandler(http.server.SimpleHTTPRequestHandler):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, directory="output", **kwargs)
    def log_message(self, format, *args):
        pass

def main():
    build()
    watcher_thread = threading.Thread(target=watch_and_rebuild, daemon=True)
    watcher_thread.start()
    server = http.server.HTTPServer(("", PORT), QuietHandler)
    url = f"http://localhost:{PORT}"
    print(f"🌐 本地预览：{url}")
    webbrowser.open(url)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\n👋 服务器已停止")

if __name__ == "__main__":
    main()

本地构建和预览
运行命令：

python build.py
python serve.py   # 自动打开浏览器，修改文件自动刷新
发布到线上 运行命令： 
git add .
git commit -m "新文章"
git push
等待 GitHub Actions 自动部署（约 1-2 分钟），线上站点即更新。 九、常见路径问题说明 本项目在 GitHub Pages 上部署时，所有链接均加了仓库名前缀 /damo-/。如果你自己新建了不同名字的仓库，需要全局替换 /damo-/ 为 /你的仓库名/。 具体涉及文件： build.py 中的 url 字段 templates/*.html 中的所有 href="/damo-/... static/style.css 不需要改，但 CSS 中的图片路径如果存在也需加上仓库名前缀。 恭喜！你已经拥有一个完全自主、可读性极强、支持搜索和本地预览的静态博客系统。
