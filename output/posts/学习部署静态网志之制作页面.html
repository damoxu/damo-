点击展开 build.py 完整代码（约 150 行） python # build.py # 作用：把 content/ 里的 Markdown 文件，转成 output/ 里的 HTML 文件 import os import shutil import markdown from jinja2 import Environment, FileSystemLoader # ───────────────────────────────────────── # 1. 配置：整个项目只有这里需要改 # ───────────────────────────────────────── BLOG_TITLE = "我的博客" AUTHOR_NAME = "damoxu" DIR_CONTENT = "content" # Markdown 文章放这里 DIR_TEMPLATES = "templates" # HTML 模板放这里 DIR_STATIC = "static" # CSS / 图片放这里 DIR_OUTPUT = "output" # 生成结果放这里 # ───────────────────────────────────────── # 2. 工具函数：每个函数只做一件事 # ───────────────────────────────────────── def read_markdown_file(filepath): """ 读取一个 Markdown 文件，返回： - meta : 文章头部信息（标题、日期、标签） - html : 正文转成的 HTML 字符串 """ with open(filepath, encoding="utf-8") as f: raw_text = f.read() md = markdown.Markdown(extensions=["meta", "fenced_code", "tables"]) html_body = md.convert(raw_text) meta = {key: values[0] for key, values in md.Meta.items()} return meta, html_body def collect_all_posts(posts_dir): posts = [] for filename in sorted(os.listdir(posts_dir), reverse=True): if not filename.endswith(".md"): continue filepath = os.path.join(posts_dir, filename) meta, html_body = read_markdown_file(filepath) post = { "slug" : filename.replace(".md", ""), "title" : meta.get("title", "无标题"), "date" : meta.get("date", "未知日期"), "tags" : meta.get("tags", "").split(","), "body" : html_body, "url" : f"/damo-/posts/{filename.replace('.md', '.html')}", } posts.append(post) return posts def collect_all_works(works_dir): works = [] for filename in sorted(os.listdir(works_dir)): if not filename.endswith(".md"): continue filepath = os.path.join(works_dir, filename) meta, html_body = read_markdown_file(filepath) work = { "slug" : filename.replace(".md", ""), "title" : meta.get("title", "无标题"), "desc" : meta.get("desc", ""), "link" : meta.get("link", "#"), "body" : html_body, "url" : f"/damo-/works/{filename.replace('.md', '.html')}", } works.append(work) return works def setup_jinja(templates_dir): return Environment(loader=FileSystemLoader(templates_dir)) def render_page(env, template_name, output_path, **data): template = env.get_template(template_name) html = template.render(**data, blog_title=BLOG_TITLE, author=AUTHOR_NAME) os.makedirs(os.path.dirname(output_path), exist_ok=True) with open(output_path, "w", encoding="utf-8") as f: f.write(html) print(f" ✓ 生成：{output_path}") def clean_output_dir(): if os.path.exists(DIR_OUTPUT): shutil.rmtree(DIR_OUTPUT) os.makedirs(DIR_OUTPUT) def copy_static_files(): src = DIR_STATIC dst = os.path.join(DIR_OUTPUT, "static") shutil.copytree(src, dst) print(f" ✓ 复制静态文件：{src} → {dst}") def build_search_index(posts, works, output_dir): import json import re index = [] for post in posts: plain_text = re.sub(r"<[^>]+>", "", post["body"]) index.append({ "title" : post["title"], "date" : post["date"], "tags" : post["tags"], "url" : post["url"], "body" : plain_text[:500], "type" : "post", }) for work in works: plain_text = re.sub(r"<[^>]+>", "", work["body"]) index.append({ "title" : work["title"], "desc" : work["desc"], "url" : work["url"], "body" : plain_text[:500], "type" : "work", }) index_path = os.path.join(output_dir, "search-index.json") with open(index_path, "w", encoding="utf-8") as f: json.dump(index, f, ensure_ascii=False, indent=2) print(f" ✓ 生成搜索索引：{index_path}") def build(): print("\n🔨 开始构建...\n") clean_output_dir() copy_static_files() posts = collect_all_posts(os.path.join(DIR_CONTENT, "posts")) works = collect_all_works(os.path.join(DIR_CONTENT, "works")) env = setup_jinja(DIR_TEMPLATES) print("\n📝 生成文章页面：") for post in posts: output_path = os.path.join(DIR_OUTPUT, post["url"].lstrip("/")) render_page(env, "post.html", output_path, post=post) print("\n🎨 生成作品页面：") for work in works: output_path = os.path.join(DIR_OUTPUT, work["url"].lstrip("/")) render_page(env, "work.html", output_path, work=work) print("\n🏠 生成首页：") render_page(env, "index.html", os.path.join(DIR_OUTPUT, "index.html"), posts=posts, works=works) print("\n🔍 生成搜索索引：") build_search_index(posts, works, DIR_OUTPUT) print("\n🔍 生成搜索页：") render_page(env, "search.html", os.path.join(DIR_OUTPUT, "search.html"), posts=posts, works=works) print("\n✅ 构建完成！\n") if __name__ == "__main__": build()

    六、本地预览服务器 serve.py python

    serve.py

    使用方式： python serve.py

    import os import time import threading import http.server import webbrowser from build import build

    PORT = 8000 WATCH_DIRS = ["content", "templates", "static"] CHECK_INTERVAL = 1

    def get_all_files_with_mtime(dirs): files = {} for directory in dirs: if not os.path.exists(directory): continue for root, _, filenames in os.walk(directory): for filename in filenames: filepath = os.path.join(root, filename) files[filepath] = os.path.getmtime(filepath) return files

    def has_any_file_changed(old, new): if set(old.keys()) != set(new.keys()): return True for filepath, mtime in new.items(): if old.get(filepath) != mtime: return True return False

    def watch_and_rebuild(): print(f"👀 监听文件变化中：{WATCH_DIRS}") snapshot = get_all_files_with_mtime(WATCH_DIRS) while True: time.sleep(CHECK_INTERVAL) new_snapshot = get_all_files_with_mtime(WATCH_DIRS) if has_any_file_changed(snapshot, new_snapshot): print("\n📄 检测到文件变化，重新构建...") build() snapshot = new_snapshot

    class QuietHandler(http.server.SimpleHTTPRequestHandler): def init(self, args, kwargs): super().init(args, directory="output", *kwargs) def log_message(self, format, args): pass

    def main(): build() watcher_thread = threading.Thread(target=watch_and_rebuild, daemon=True) watcher_thread.start() server = http.server.HTTPServer(("", PORT), QuietHandler) url = f"http://localhost:{PORT}" print(f"🌐 本地预览：{url}") webbrowser.open(url) try: server.serve_forever() except KeyboardInterrupt: print("\n👋 服务器已停止")

    if name == "main": main() 七、部署到 GitHub Pages 1. 初始化 Git 仓库 运行命令：

    bash git init git add . git commit -m "first commit" 2. 创建 .gitignore bash echo -e "pycache/\n*.pyc\n.DS_Store" > .gitignore git add .gitignore git commit -m "add gitignore" 3. 在 GitHub 上新建仓库 登录 GitHub，点击 New repository

    仓库名：damo-（可自定义）

    设为 Public

    不要勾选初始化选项

    复制仓库地址（例如 https://github.com/damoxu/damo-.git）

    1. 关联并推送 运行命令：

    bash git remote add origin https://github.com/damoxu/damo-.git git branch -M main git push -u origin main 5. 创建 GitHub Actions 工作流 创建文件 .github/workflows/deploy.yml：

    yaml name: 构建并部署博客

    on: push: branches: [main]

    jobs: build-and-deploy: runs-on: ubuntu-latest steps: - uses: actions/checkout@v4 - name: 安装 Python uses: actions/setup-python@v5 with: python-version: "3.12" - name: 安装依赖 run: pip install markdown jinja2 - name: 构建博客 run: python build.py - name: 发布到 GitHub Pages uses: peaceiris/actions-gh-pages@v3 with: github_token: ${{ secrets.GITHUB_TOKEN }} publish_dir: ./output 推送：

    bash git add .github/workflows/deploy.yml git commit -m "add github actions deploy" git push 6. 开启 Actions 写入权限 打开仓库 → Settings → Actions → General

    滚动到 Workflow permissions，选择 Read and write permissions，保存。

    1. 设置 Pages 来源 打开仓库 → Settings → Pages

    Source → 选择 Deploy from a branch

    Branch 选择 gh-pages，目录 / (root)

    点击 Save

    等待 1-2 分钟，博客访问地址为： https://damoxu.github.io/damo-/ （将 damoxu 换成你的 GitHub 用户名）
