  <div id="overlay"></div>

  <script>
    const keywords = ['杜益', '520', '许德思', '王强', '高原', '吕顺远','1314','韩韬淮'];
    const hearts = ['❤️', '💕', '💖', '💗', '💓', '💘', '💝', '💞', '🩷'];

    let active = false;
    let timer = null;
    let burstTimer = null;

    const input = document.getElementById('searchInput');

    input.addEventListener('input', () => {
      const val = input.value;
      const matched = keywords.some(k => val.includes(k));

      if (matched && !active) {
        startHearts();
      } else if (!matched && active) {
        stopHearts();
      }
    });

    function startHearts() {
      active = true;
      burst(60);
      timer = setInterval(() => {
        for (let i = 0; i < 5; i++) {
          setTimeout(() => spawnHeart(), i * 80);
        }
      }, 300);
    }

    function stopHearts() {
      active = false;
      clearInterval(timer);
    }

    function burst(count) {
      for (let i = 0; i < count; i++) {
        setTimeout(() => spawnHeart(), i * 30);
      }
    }

    function spawnHeart() {
      const el = document.createElement('div');
      el.classList.add('heart');
      el.textContent = hearts[Math.floor(Math.random() * hearts.length)];

      const x = Math.random() * window.innerWidth;
      const y = window.innerHeight * (0.4 + Math.random() * 0.6);
      const size = 18 + Math.random() * 28;
      const dur = 2 + Math.random() * 2.5;

      el.style.left = x + 'px';
      el.style.top = y + 'px';
      el.style.fontSize = size + 'px';
      el.style.animationDuration = dur + 's';

      document.body.appendChild(el);
      setTimeout(() => el.remove(), dur * 1000);
    }
  </script>
