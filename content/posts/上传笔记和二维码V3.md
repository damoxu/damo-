  // 编码笔记内容到 URL
  function encodePost(p){
    const d={author:p.author,title:p.title,body:p.body,time:p.time};
    try{return btoa(unescape(encodeURIComponent(JSON.stringify(d))));}catch(e){return '';}
  }
  function decodePost(str){
    try{return JSON.parse(decodeURIComponent(escape(atob(str))));}catch(e){return null;}
  }

  // 检查 URL 是否是扫码进来的
  function checkUrlPost(){
    const hash=window.location.hash;
    if(!hash.startsWith('#view='))return false;
    const data=decodePost(hash.slice(6));
    if(!data)return false;
    document.getElementById('composerSection').style.display='none';
    document.getElementById('barSection').style.display='none';
    document.getElementById('feed').style.display='none';
    document.getElementById('singleView').style.display='block';

        </div>
      </div>
      ${data.title?`<div class="post-title">${escHtml(data.title)}</div>`:''}
      ${data.body?`<div class="post-body">${escHtml(data.body)}</div>`:''}

  function formatTime(iso){
    const d=new Date(iso),now=new Date(),diff=(now-d)/1000;
    if(diff<60)return '刚刚';
    if(diff<3600)return Math.floor(diff/60)+' 分钟前';
    if(diff<86400)return Math.floor(diff/3600)+' 小时前';
    return d.toLocaleDateString('zh-CN',{month:'2-digit',day:'2-digit',hour:'2-digit',minute:'2-digit'});
  }
