def read_markdown_file(filepath): """ 读取一个 Markdown 文件，返回： - meta : 文章头部信息（标题、日期、标签） - html : 正文转成的 HTML 字符串 """ with open(filepath, encoding="utf-8") as f: raw_text = f.read()

# markdown 库同时处理正文和头部元数据
md = markdown.Markdown(extensions=["meta", "fenced_code", "tables"])
html_body = md.convert(raw_text)

# md.Meta 里的值是列表，取第一个元素变成字符串
meta = {key: values[0] for key, values in md.Meta.items()}
return meta, html_body

def collect_all_posts(posts_dir): """ 扫描 content/posts/ 目录，收集所有文章信息。 返回一个列表，每项是一篇文章的字典。 """ posts = []

for filename in sorted(os.listdir(posts_dir), reverse=True):
    if not filename.endswith(".md"):
        continue  # 跳过非 Markdown 文件

    filepath = os.path.join(posts_dir, filename)
    meta, html_body = read_markdown_file(filepath)

    post = {
        "slug"  : filename.replace(".md", ""),  # 用于生成文件名
        "title" : meta.get("title", "无标题"),
        "date"  : meta.get("date",  "未知日期"),
        "tags"  : meta.get("tags",  "").split(","),
        "body"  : html_body,
        "url"   : f"posts/{filename.replace('.md', '.html')}",
    }
    posts.append(post)

return posts

def collect_all_works(works_dir): """ 扫描 content/works/ 目录，收集所有作品信息。 逻辑和 collect_all_posts 一样，单独分开方便以后扩展。 """ works = []

for filename in sorted(os.listdir(works_dir)):
    if not filename.endswith(".md"):
        continue

    filepath = os.path.join(works_dir, filename)
    meta, html_body = read_markdown_file(filepath)

    work = {
        "slug"  : filename.replace(".md", ""),
        "title" : meta.get("title", "无标题"),
        "desc"  : meta.get("desc",  ""),
        "link"  : meta.get("link",  "#"),
        "body"  : html_body,
        "url"   : f"works/{filename.replace('.md', '.html')}",
    }
    works.append(work)

return works

def setup_jinja(templates_dir): """初始化 Jinja2 模板引擎，告诉它去哪里找模板文件。""" env = Environment(loader=FileSystemLoader(templates_dir)) return env

def render_page(env, template_name, output_path, data): """ 通用渲染函数： - env : Jinja2 引擎 - template_name : 用哪个模板（如 "post.html"） - output_path : 输出到哪里（如 "output/posts/my-post.html"） - data : 传给模板的所有变量 """ template = env.get_template(template_name) html = template.render(**data, blog_title=BLOG_TITLE, author=AUTHOR_NAME)

os.makedirs(os.path.dirname(output_path), exist_ok=True)
with open(output_path, "w", encoding="utf-8") as f:
    f.write(html)

print(f"  ✓ 生成：{output_path}")

print("\n📝 生成文章页面：")

    output_path = os.path.join(DIR_OUTPUT, post["url"])
    render_page(env, "post.html", output_path, post=post)

# 步骤 5：每个作品单独生成一个 HTML 页面

render_page(env, "index.html",
            os.path.join(DIR_OUTPUT, "index.html"),
            posts=posts,
            works=works)

开启 GitHub Pages

打开 https://github.com/damoxu/damo- 点顶部 Settings 左侧菜单找 Pages 在 Source 下选 GitHub Actions

创建自动部署脚本 每次你 git push，GitHub 会自动运行这个脚本，把 output/ 发布成网站。 bashmkdir -p .github/workflows cat > .github/workflows/deploy.yml 粘贴以下内容，Ctrl + D 保存： yaml# .github/workflows/deploy.yml

作用：每次推送到 main 分支，自动构建并发布博客

name: 构建并部署博客

on: push: branches: [main] # 只监听 main 分支的推送

jobs: build-and-deploy: runs-on: ubuntu-latest

steps:
  # 1. 拉取代码
  - name: 拉取代码
    uses: actions/checkout@v4

  # 2. 安装 Python
  - name: 安装 Python
    uses: actions/setup-python@v5
    with:
      python-version: "3.12"

  # 3. 安装依赖
  - name: 安装依赖
    run: pip install markdown jinja2

  # 4. 运行构建脚本，生成 output/
  - name: 构建博客
    run: python build.py

  # 5. 把 output/ 目录发布到 GitHub Pages
  - name: 发布到 GitHub Pages
    uses: peaceiris/actions-gh-pages@v3
    with:
      github_token: ${{ secrets.GITHUB_TOKEN }}
      publish_dir: ./output

然后推送到 GitHub： bashgit add . git commit -m "add github actions deploy" git push 推送后告诉我，我们去 GitHub 上看部署是否成功。You said: git push(ct) [damoxu@whshare-ccs-cli-2 damo许个人空间]$ git commit -m "add github actions deploy"git push(ct) [damoxu@whshare-ccs-cli-2 damo许个人空间]$ git commit -m "add github actions deploy" [main 7ef5623] add github actions deploy 2 files changed, 41 insertions(+) create mode 100644 .github/workflows/deploy.yml create mode 100644 .gitignore (ct) [damoxu@whshare-ccs-cli-2 damo许个人空间]$ git push16:17Claude responded: 好，现在去 GitHub 看部署状态：好，现在去 GitHub 看部署状态：
